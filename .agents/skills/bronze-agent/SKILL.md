# Source to Bronze - Complete Pipeline Architecture
This document outlines the **complete suite of Python/PySpark scripts** that the Agent will generate and upload to the GCS bucket (`codebase_etl_pipeline/<project_id>_pipeline/`). 

Together, these files form the fully automated Source-to-Bronze framework.

## 1. `config/config.yaml`
**Purpose**: Stores the dynamic configuration for the project (Project ID, source paths, target datasets, required columns, high-watermark definitions, data quality rules, and anomaly detection logic).

**Inputs to the YAML Generation Prompt:**
The agent dynamically injects the following inputs from the user's project state into the prompt before sending it to the LLM:
1. `gcp_project_id`: The target Google Cloud Project ID.
2. `selected_tables`: A JSON object/list of which tables the user wants to ingest.
3. `source_type`: e.g., "GCS", "BigQuery", "S3".
4. `source_path`: The raw data source URI (e.g., `gs://bucket-name/folder`).
5. `file_formats`: The format of the raw data (e.g., `CSV`, `JSON`, `PARQUET`).
6. `watermark`: The column name used for incremental loading (e.g., `updated_at`).
7. `dq_rules_context`: A formatted string containing the SQL Data Quality rules generated by the DQ Agent.
8. `anomalies_context`: A formatted string containing the Anomaly Detection rules generated by the Anomaly Agent.
9. `generated_doc`: The Markdown schema/data model created by the EDA/Business Agent.

## 2. `src/utils/config_loader.py`
**Purpose**: A lightweight utility to load the YAML configuration safely across all scripts.
```python
import yaml
import os

def load_config(file_path="config/config.yaml"):
    if not os.path.exists(file_path) and os.path.exists(os.path.basename(file_path)):
        file_path = os.path.basename(file_path)
    with open(file_path, "r") as f:
        return yaml.safe_load(f)
```

## 3. `src/utils/alert.py`
**Purpose**: Handles sending alerts (HIGH, WARNING) to Slack when schema evolution detects anomalies or missing required columns.
```python
import os
import requests
import json

def publish_alert(severity, payload, project_id):
    print(f"[{severity}] ALERT for Project {project_id}: {payload}")
    
    slack_webhook_url = os.environ.get("SLACK_WEBHOOK_URL")
    if not slack_webhook_url:
        print("Warning: SLACK_WEBHOOK_URL not set. Skipping Slack alert.")
        return

    # Format the message for Slack
    message_text = f"*{severity} ALERT* (Project: {project_id})\n"
    for k, v in payload.items():
        message_text += f"• *{k}*: {v}\n"

    slack_data = {"text": message_text}
    
    try:
        response = requests.post(
            slack_webhook_url, 
            data=json.dumps(slack_data),
            headers={'Content-Type': 'application/json'}
        )
        if response.status_code != 200:
            print(f"Failed to send Slack alert. Status Code: {response.status_code}")
    except Exception as e:
        print(f"Error sending Slack alert: {e}")
```

## 4. `src/readers/file_reader.py`
**Purpose**: Handles the initial connection to the raw source data (GCS/AWS/DBs).
```python
from pyspark.sql import SparkSession
from google.cloud import storage
import datetime

def read_source_data(spark: SparkSession, table_config: dict, project_id: str = None, max_wm=None):
    """
    Reads data based on the format specified in the config.
    """
    file_format = table_config.get('format', '').lower()
    
    if file_format in ['bigquery', 'table']:
        dataset_id = table_config.get('dataset_id')
        table_name = table_config.get('table_name')
        fq_table = f"{project_id}.{dataset_id}.{table_name}" if project_id else f"{dataset_id}.{table_name}"
        df = spark.read.format("bigquery").option("table", fq_table).load()
        if max_wm and table_config.get('watermark_column'):
            from pyspark.sql.functions import col, lit
            df = df.filter(col(table_config['watermark_column']) > lit(max_wm))
        return df
        
    # Handle GCS sources
    source_path = table_config.get('source_path')
    bucket_name = table_config.get('bucket_name')
    if not source_path and bucket_name:
        source_path = f"gs://{bucket_name}/{table_config.get('path', '')}"

    read_path = source_path
    


    if file_format == 'csv':
        # To handle schema evolution (missing columns) in CSV files correctly,
        # we must read them individually and union them by name.
        if bucket_name:
            client = storage.Client()
            bucket = client.bucket(bucket_name)
            prefix = table_config.get('path', '')
            blobs = list(bucket.list_blobs(prefix=prefix))
            csv_blobs = [b for b in blobs if b.name.endswith('.csv')]
            
            if not csv_blobs:
                return None
                
            # Sort blobs by updated time descending to find the newest file
            csv_blobs.sort(key=lambda b: b.updated, reverse=True)
            newest_file = f"gs://{bucket_name}/{csv_blobs[0].name}"
            
            # Read newest file to get the "source of truth" schema for this run
            newest_df = spark.read.option("header", "true").option("inferSchema", "true").csv(newest_file)
            target_columns = newest_df.columns
            
            from functools import reduce
            csv_files = [f"gs://{bucket_name}/{b.name}" for b in csv_blobs]
            dfs = [spark.read.option("header", "true").option("inferSchema", "true").csv(f) for f in csv_files]
            
            # Union all files together, padding missing columns with nulls
            merged_df = reduce(lambda df1, df2: df1.unionByName(df2, allowMissingColumns=True), dfs)
            
            # Force the final dataframe to exactly match the newest file's schema!
            # This automatically drops removed columns and includes newly added columns.
            return merged_df.select(*target_columns)
        else:
            return spark.read.option("header", "true").option("inferSchema", "true").csv(read_path)
    elif file_format in ['json', 'jsonl']:
        return spark.read.json(read_path)
    elif file_format == 'parquet':
        return spark.read.parquet(read_path)
    else:
        raise ValueError(f"Unsupported format: {file_format}")
```

## 5. `src/bronze/bronze_processor.py` (The Core Engine)
**Purpose**: The main PySpark orchestrator that handles Incremental loads, Universal GCS Backups, Schema Evolution, and Truncate-and-Load to BigQuery Bronze.

**Prompt Rules Used for Generation:**
When generating this PySpark code, the LLM will be strictly instructed to follow these rules:
1. **Ensure Dataset Exists:** The code MUST explicitly ensure the BigQuery Bronze dataset exists, creating it if it is completely missing on the first run.
2. **Incremental Logic & Table Creation:** The code MUST attempt to query the BigQuery target table for the `MAX` watermark. If the table does NOT exist, it must safely catch the exception and perform a **full initial load**. If the table DOES exist, it must perform an **incremental load** by filtering for new rows.
3. **Transient Staging Overwrite:** Write the schema-enforced filtered DataFrame back to BigQuery using `.mode("overwrite")` so that Bronze only holds the most recent batch. You MUST configure the Spark write with `.option("temporaryGcsBucket", config['gcs_backup_bucket'])`. This guarantees the BigQuery table is seamlessly created via the Load API on the first run, and fully dropped/overwritten on subsequent runs.
4. **Natural Schema Inference**: The PySpark job MUST NOT pad missing columns. It should just use the exact schema inferred from the raw source file.
5. **Universal GCS Backup (Post-Load)**: AFTER the data is successfully written to the Bronze BigQuery table, the exact same batch is appended to the GCS Backup location as Parquet files to maintain true history.

```python
import sys
from pyspark.sql import SparkSession
from pyspark.sql.functions import col, lit, max as spark_max, current_timestamp
from pyspark.sql.types import StringType
from google.cloud import bigquery
from src.utils.config_loader import load_config
from src.readers.file_reader import read_source_data
from src.utils.alert import publish_alert

def process_bronze():
    # 1. Initialize Spark & BQ
    spark = SparkSession.builder \
        .appName("SourceToBronze_Ingestion") \
        .getOrCreate()
    bq_client = bigquery.Client()
    
    # 2. Load Config
    config = load_config()
    project_id = config['project_id']
    
    # 3. Process Each Table
    for table in config['source']['tables']:
        table_name = table['name']
        target_dataset_id = table['target_dataset_id']
        target_bq_table = f"{project_id}.{target_dataset_id}.{table['target_table_name']}"
        gcs_backup_path = f"gs://{config['gcs_backup_bucket']}/{project_id}/backup/{table_name}/"
        
        # Ensure target dataset exists
        dataset_ref = bigquery.Dataset(f"{project_id}.{target_dataset_id}")
        bq_client.create_dataset(dataset_ref, exists_ok=True)
        
        watermark_col = table.get('watermark_column')
        
        # A. FETCH MAX WATERMARK
        max_wm = None
        if watermark_col:
            try:
                query = f"SELECT MAX({watermark_col}) as max_wm FROM `{target_bq_table}`"
                query_job = bq_client.query(query)
                results = query_job.result()
                for row in results:
                    max_wm = row.max_wm
            except Exception as e:
                print(f"Table {target_bq_table} not found or error querying BQ: {e}. Doing full load.")
                
        # B. READ SOURCE DATA
        raw_df = read_source_data(spark, table, project_id, max_wm)
        
        # C. APPLY INCREMENTAL FILTER FOR GCS DATA
        if raw_df:
            incremental_df = raw_df
            if max_wm:
                incremental_df = raw_df.filter(col(watermark_col) > lit(max_wm))
        else:
            incremental_df = None
                
        if incremental_df is None or incremental_df.count() == 0:
            print(f"No new data found for {table_name}. Skipping.")
            continue
            
        # Add Audit Column
        incremental_df = incremental_df.withColumn("bronze_ingestion_timestamp", current_timestamp())
        
        # C. PREPARE LATEST BATCH FOR BIGQUERY
        final_df = incremental_df
            
        # D. REPLACE BIGQUERY TABLE (TRANSIENT STAGING)
        # Forcefully delete old table to bypass BigQuery column shrink protections
        bq_client.delete_table(target_bq_table, not_found_ok=True)
        # Overwrite BigQuery table so it only contains the latest incremental batch
        final_df.write \
            .format("bigquery") \
            .option("table", target_bq_table) \
            .option("temporaryGcsBucket", config['gcs_backup_bucket']) \
            .mode("overwrite") \
            .save()
            
        # E. WRITE NEW DATA TO GCS BACKUP
        final_df.write.mode("append").format("parquet").save(gcs_backup_path)

if __name__ == "__main__":
    process_bronze()
```
